// ---------- Görev Fonksiyonları ----------
void TaskCore0(void *pvParameters)
{
  //------ RGB LEDİ AYARLA ------
  while (true)
  {
    if (digitalRead(KAPAK_PIN) == HIGH)
    {
      for (int i = 0; i < LED_COUNT; i++)
      {
        strip.setPixelColor(i, 255, 255, 255);
      }
    }
    else
    {
      for (int i = 0; i < LED_COUNT; i++)
      {
        strip.setPixelColor(i, 0, 0, 0);
      }
      bilgileriYazdir();
    }
    strip.show();

    //------RGB LEDİ AYARLA Bitis----
    //

    // Serial.println("Core 0: LED veya sensör görevi...");
    if (analogRead(FIRCA1_PIN) < 1000 && isPlaying == false)
    { // Buton1'e basıldıysa
      Serial.println("Firca 1 cikarildi");
      fircaId = "Firca1";
      klasor = "/1";
      isPlaying = true;
      delay(100);
    }
    else if (analogRead(FIRCA2_PIN) < 1000 && isPlaying == false)
    {
      Serial.println("FIRCA 2 cikarildi");
      fircaId = "Firca2";
      klasor = "/2";
      isPlaying = true;
      delay(100);
    }
    else if (analogRead(FIRCA1_PIN) > 1000 && analogRead(FIRCA2_PIN) > 1000 && analogRead(FIRCA3_PIN) > 1000 && analogRead(FIRCA4_PIN) > 1000 && analogRead(FIRCA5_PIN) > 1000)
    {
      Serial.println("Fircalar yerinde");

      if (calis)
      {
        Serial.println("Resetle!");
        calis = false;
        resetTime();
        delay(100);
        lc.clearDisplay(MATRIX_A);
        lc.clearDisplay(MATRIX_B);
      }
      fircaId = "";
      klasor = "";
      MuzikDurdurFnc();
    }
    // TODO: Veriyi gönder sonra müzik çal

    if (digitalRead(BTNSTOP_PIN) == HIGH)
    {
      Serial.println("Muzik Durduruldu");
      MuzikDurdurFnc();
    }
    if (mp3.isRunning())
    {
      mp3.loop();
    }
    else
    {
      if (isPlaying)
      {
        Serial.println("Muzik yeniden oynatılıyor");
        const char *charUrl = klasor.c_str();
        if (klasor != "")
        {
          playRandomSongFromFolder(charUrl);
        }
      }

      delay(100);
    }

    vTaskDelay(50 / portTICK_PERIOD_MS);
  }
}

void TaskCore1(void *pvParameters)
{
  while (true)
  {
    // Serial.println("Core 1: Ağ, ses, vs. görevi...");
    /*Kum saati baslangıcı*/
    if (digitalRead(BTNPLAY_PIN) == HIGH)
    {
      Serial.println("Baslat!");
      for (byte i = 0; i < 2; i++)
      {
        lc.shutdown(i, false);
        lc.setIntensity(i, 2);
      }
      resetTime();
      if (fircaId != "" && wifiStatus == true)
      {
        while (!DatayiGonder(cihazId, fircaId))
        {
          Serial.println("Veri gonderiliyor...");
        }
        isPlaying = true;
      }
      else
      {
        Serial.println("Wifi baglantisi yok");
      }

      if (millis() - lastDebounce > DEBOUNCE_THRESHOLD)
      {
        calis = !calis;
        Serial.println(calis);
        lastDebounce = millis();
      }
    }
    if (digitalRead(BTNREFRESH_PIN) == HIGH)
    {
      Serial.println("Resetle!");
      for (byte i = 0; i < 2; i++)
      {
        lc.shutdown(i, false);
        lc.setIntensity(i, 2);
      }
      calis = false;
      resetTime();
    }
    if (calis)
    {
      delay(DELAY_FRAME);
      gravity = getGravity();
      lc.setRotation((ROTATION_OFFSET + gravity) % 360);
      // Serial.println(gravity);
      bool moved = updateMatrix();
      bool dropped = dropParticle();
    }
    /*Kum saati bitis*/
    vTaskDelay(50 / portTICK_PERIOD_MS);
  }
}
